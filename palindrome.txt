class Node {
    int data;
    Node next;

    Node(int data) {
        this.data = data;
        this.next = null;
    }
}

public class Main {

    static Node head;

    // Insert at end
    static void insert(int data) {
        Node newNode = new Node(data);

        if (head == null) {
            head = newNode;
            return;
        }

        Node temp = head;

        while (temp.next != null) {
            temp = temp.next;
        }

        temp.next = newNode;
    }

    // Check palindrome
    static boolean isPalindrome() {

        Node temp = head;
        int count = 0;

        // Find length
        while (temp != null) {
            count++;
            temp = temp.next;
        }

        int[] arr = new int[count];

        temp = head;
        int i = 0;

        // Store elements in array
        while (temp != null) {
            arr[i] = temp.data;
            i++;
            temp = temp.next;
        }

        // Compare from both ends
        for (i = 0; i < count / 2; i++) {
            if (arr[i] != arr[count - 1 - i]) {
                return false;
            }
        }

        return true;
    }

    public static void main(String[] args) {

        insert(1);
        insert(2);
        insert(3);
        insert(2);
        insert(1);

        if (isPalindrome()) {
            System.out.println("Palindrome");
        } else {
            System.out.println("Not Palindrome");
        }
    }
}
